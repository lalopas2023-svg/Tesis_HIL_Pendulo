// ============================================================
// CONTROLADOR + PLANTA HIL (CORREGIDO Y VERIFICADO)
// ============================================================

#include "F28x_Project.h"
#include <stdio.h>
#include <string.h>

// --- DEFINICIONES DE CONTROLADOR ---
#define PI              3.14159265359f
#define Ts              0.00001f        // 10 us   
#define K1              1.0f
#define K2              0.15f
#define U_MIN           -10.0f
#define U_MAX           10.0f
#define THETA_MIN       -1.570796f      
#define THETA_MAX       1.570796f       
#define THETA_RANGE     (THETA_MAX - THETA_MIN)

// --- DEFINICIONES DE PLANTA ---
#define g               9.81f
#define L               0.3f
#define m               0.1f
#define J               (m * L * L)
#define invJ            (1.0f / J)
#define gL              (g / L)

// --- VARIABLES GLOBALES ---
volatile float theta = 15.0f;             
volatile float theta_dot = 0.0f;
volatile float theta_prev = 0.0f;
volatile float u = 0.0f;
volatile float theta_measured = 0.0f;
volatile float theta_dot_filt = 0.0f;
volatile int flag_enviar = 0;
volatile int contador_serial = 0;
char buffer_serial[50]; 

__interrupt void cpu_timer0_isr(void);

// --- FUNCIONES SERIALES (SCI-A) ---
void InitSciaGpio_LaunchPad(void) {
    EALLOW;
    // RX conectado al USB en GPIO43
    GpioCtrlRegs.GPBPUD.bit.GPIO43 = 0;   // Activar pull-up
    GpioCtrlRegs.GPBQSEL1.bit.GPIO43 = 3; // Asíncrono
    GpioCtrlRegs.GPBGMUX1.bit.GPIO43 = 3; // Mux Group 3
    GpioCtrlRegs.GPBMUX1.bit.GPIO43 = 3;  // Mux 3 (Total = 15 -> SCIRXDA)

    // TX conectado al USB en GPIO42
    GpioCtrlRegs.GPBPUD.bit.GPIO42 = 0;   // Activar pull-up
    GpioCtrlRegs.GPBGMUX1.bit.GPIO42 = 3; // Mux Group 3
    GpioCtrlRegs.GPBMUX1.bit.GPIO42 = 3;  // Mux 3 (Total = 15 -> SCITXDA)
    EDIS;
}

void InitScia_Config(void) {
    EALLOW;
    CpuSysRegs.PCLKCR7.bit.SCI_A = 1; // Habilitar reloj para SCI-A
    EDIS;

    SciaRegs.SCICCR.all = 0x0007;   // 1 stop bit, sin paridad, 8 bits de datos
    SciaRegs.SCICTL1.all = 0x0003;  // Habilitar TX y RX
    SciaRegs.SCICTL2.all = 0x0003;
    SciaRegs.SCIHBAUD.all = 0x0000;
    SciaRegs.SCILBAUD.all = 53;     // 115200 baudios (LSPCLK = 50MHz)
    SciaRegs.SCICTL1.all = 0x0023;  // Salir de modo reset
}

void SciSendString(char *msg) {
    int i = 0;
    while(msg[i] != '\0') {
        while (SciaRegs.SCICTL2.bit.TXRDY == 0) {} 
        SciaRegs.SCITXBUF.all = msg[i];
        i++;
    }
}

void main(void) {
    volatile long delay_i; 

    InitSysCtrl();
    InitGpio();
    DINT;
    InitPieCtrl();
    IER = 0x0000;
    IFR = 0x0000;
    InitPieVectTable();

    // CONFIGURACIÓN SERIAL
    InitSciaGpio_LaunchPad();
    InitScia_Config();

    // CONFIGURACIÓN ADC
    EALLOW;
    CpuSysRegs.PCLKCR13.bit.ADC_A = 1; 
    AdcaRegs.ADCCTL2.bit.PRESCALE = 6; 
    AdcaRegs.ADCCTL1.bit.ADCPWDNZ = 1;
    AdcaRegs.ADCCTL1.bit.INTPULSEPOS = 1;
    EDIS;
    
    for(delay_i = 0; delay_i < 200000; delay_i++){}

    EALLOW;
    AdcaRegs.ADCSOC0CTL.bit.CHSEL = 0;
    AdcaRegs.ADCSOC0CTL.bit.ACQPS = 14;
    AdcaRegs.ADCSOC0CTL.bit.TRIGSEL = 0;
    AdcaRegs.ADCINTSEL1N2.bit.INT1SEL = 0;
    AdcaRegs.ADCINTSEL1N2.bit.INT1E = 1;
    EDIS;

    // CONFIGURACIÓN DAC
    EALLOW;
    CpuSysRegs.PCLKCR16.bit.DAC_B = 1; 
    DacbRegs.DACCTL.bit.DACREFSEL = 1;      
    DacbRegs.DACCTL.bit.LOADMODE = 0;       
    DacbRegs.DACOUTEN.bit.DACOUTEN = 1;     
    DacbRegs.DACVALS.all = 2048;            
    EDIS;

    for(delay_i = 0; delay_i < 10000; delay_i++){}

    // CONFIGURACIÓN TIMER 0 (1ms exacto)
    InitCpuTimers();
    ConfigCpuTimer(&CpuTimer0, 200, 10); // 10us 

    EALLOW;
    PieVectTable.TIMER0_INT = &cpu_timer0_isr;
    EDIS;

    PieCtrlRegs.PIEIER1.bit.INTx7 = 1;
    IER |= M_INT1;
    EINT;
    ERTM;
    CpuTimer0Regs.TCR.bit.TSS = 0; // Arranca el Timer

    // ==========================================================
    // BUCLE PRINCIPAL (Solo se envía theta)
    // ==========================================================
    for(;;) { 
        if(flag_enviar == 1) {
            int theta_scaled = (int)(theta * 1000.0f);
            
            int val, d_idx;
            char digits[8];

            // 1. Enviar theta_scaled
            val = theta_scaled;
            if (val < 0) {
                while (SciaRegs.SCICTL2.bit.TXRDY == 0) {} SciaRegs.SCITXBUF.all = '-';
                val = -val;
            }
            d_idx = 0;
            do {
                digits[d_idx++] = (val % 10) + '0';
                val /= 10;
            } while (val > 0);
            while (d_idx > 0) {
                d_idx--;
                while (SciaRegs.SCICTL2.bit.TXRDY == 0) {}
                SciaRegs.SCITXBUF.all = digits[d_idx];
            }

            // 2. Salto de línea (\r\n) para cerrar la trama
            while (SciaRegs.SCICTL2.bit.TXRDY == 0) {} SciaRegs.SCITXBUF.all = '\r';
            while (SciaRegs.SCICTL2.bit.TXRDY == 0) {} SciaRegs.SCITXBUF.all = '\n';
            
            flag_enviar = 0;
        }
    }
}

__interrupt void cpu_timer0_isr(void) {
    
    AdcaRegs.ADCINTFLGCLR.bit.ADCINT1 = 1; 
    AdcaRegs.ADCSOCFRC1.bit.SOC0 = 1;      
    
    volatile int timeout = 1000;
    while((AdcaRegs.ADCINTFLG.bit.ADCINT1 == 0) && (timeout > 0)) {
        timeout--;
    }
    AdcaRegs.ADCINTFLGCLR.bit.ADCINT1 = 1; 

    uint16_t adc_raw = AdcaResultRegs.ADCRESULT0;
    theta_measured = THETA_MIN + ((float)adc_raw / 4095.0f) * THETA_RANGE;

    float theta_dot_meas = (theta_measured - theta_prev) / Ts;
    
    // Filtro pasa-bajas para la derivada (suaviza los picos de ruido)
    theta_dot_filt = (0.95f * theta_dot_filt) + (0.05f * theta_dot_meas);
    
    // El controlador ahora usa la velocidad filtrada
    u = -(K1 * theta_measured) - (K2 * theta_dot_filt);
    
    if (u > U_MAX) u = U_MAX;
    if (u < U_MIN) u = U_MIN;

    theta_prev = theta_measured;

    float theta_dot_new = theta_dot + Ts * (gL * theta + invJ * u);
    float theta_new = theta + Ts * theta_dot;
    
    if (theta_new > THETA_MAX) theta_new = THETA_MAX;
    if (theta_new < THETA_MIN) theta_new = THETA_MIN;

    theta = theta_new;
    theta_dot = theta_dot_new;

    float theta_frac = (theta - THETA_MIN) / THETA_RANGE;
    if(theta_frac < 0.0f) theta_frac = 0.0f;
    if(theta_frac > 1.0f) theta_frac = 1.0f;
    DacbRegs.DACVALS.all = (uint16_t)(theta_frac * 4095.0f);

    // --- ACTIVAR BANDERA CADA 10ms PARA EL ENVÍO SERIAL ---
    contador_serial++;
    if(contador_serial >= 1000) { 
        flag_enviar = 1;
        contador_serial = 0;
    }

    PieCtrlRegs.PIEACK.all = PIEACK_GROUP1; 
}

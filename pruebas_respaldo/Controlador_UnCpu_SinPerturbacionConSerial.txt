// ============================================================
// CONTROLADOR + PLANTA HIL (CORREGIDO PARA PRUEBAS)
// Sin perturbación, con derivada lenta y protocolo completo
// ============================================================

#include "F28x_Project.h"
#include <stdio.h>
#include <string.h>

// --- DEFINICIONES DE CONTROLADOR ---
#define PI              3.14159265359f
#define Ts              0.00001f        // 10 us (período de simulación de la planta)
#define KP              5.0f
#define KI              2.0f
#define KD              0.15f
#define U_MIN           -10.0f
#define U_MAX           10.0f
#define THETA_MIN       -1.570796f
#define THETA_MAX       1.570796f
#define THETA_RANGE     (THETA_MAX - THETA_MIN)

// --- DEFINICIONES DE PLANTA ---
#define g               9.81f
#define L               0.3f
#define m               0.1f
#define b_fric          0.005f
#define J               (m * L * L)
#define invJ            (1.0f / J)
#define gL              (g / L)

// --- VARIABLES GLOBALES ---
volatile float theta = 0.1f;            // CORREGIDO: Iniciar dentro del rango físico
volatile float theta_dot = 0.0f;
volatile float theta_prev_slow = 0.0f;
volatile float theta_dot_meas = 0.0f;
volatile float theta_dot_filt = 0.0f;
volatile float u = 0.0f;
volatile float theta_measured = 0.0f;
volatile float error_integral = 0.0f;
volatile int   contador_derivada = 0;
volatile int   flag_enviar = 0;
volatile int   contador_serial = 0;
char buffer_serial[50];

__interrupt void cpu_timer0_isr(void);

// --- FUNCIONES SERIALES (SCI-A) ---
void InitSciaGpio_LaunchPad(void) {
    EALLOW;
    // RX -> GPIO43 (SCIRXDA)
    GpioCtrlRegs.GPBPUD.bit.GPIO43 = 0;
    GpioCtrlRegs.GPBQSEL1.bit.GPIO43 = 3;
    GpioCtrlRegs.GPBGMUX1.bit.GPIO43 = 3;
    GpioCtrlRegs.GPBMUX1.bit.GPIO43 = 3;
    // TX -> GPIO42 (SCITXDA)
    GpioCtrlRegs.GPBPUD.bit.GPIO42 = 0;
    GpioCtrlRegs.GPBGMUX1.bit.GPIO42 = 3;
    GpioCtrlRegs.GPBMUX1.bit.GPIO42 = 3;
    EDIS;
}

void InitScia_Config(void) {
    EALLOW;
    CpuSysRegs.PCLKCR7.bit.SCI_A = 1; // Habilitar reloj para SCI-A
    EDIS;

    SciaRegs.SCICCR.all = 0x0007;   // 1 stop bit, sin paridad, 8 bits de datos
    SciaRegs.SCICTL1.all = 0x0003;  // Habilitar TX y RX
    SciaRegs.SCICTL2.all = 0x0003;
    SciaRegs.SCIHBAUD.all = 0x0000;
    // CORREGIDO: Baud rate real a 115200 bps (LSPCLK=25MHz)
    // Baud = 25,000,000 / ((27+1)*8) = 115,740 bps (error 0.47%)
    SciaRegs.SCILBAUD.all = 27;
    SciaRegs.SCICTL1.all = 0x0023;  // Salir de modo reset
}

void main(void) {
    volatile long delay_i;

    InitSysCtrl();
    InitGpio();
    DINT;
    InitPieCtrl();
    IER = 0x0000;
    IFR = 0x0000;
    InitPieVectTable();

    // CONFIGURACIÓN SERIAL
    InitSciaGpio_LaunchPad();
    InitScia_Config();

    // CONFIGURACIÓN ADC
    EALLOW;
    CpuSysRegs.PCLKCR13.bit.ADC_A = 1;
    AdcaRegs.ADCCTL2.bit.PRESCALE = 6;
    AdcaRegs.ADCCTL1.bit.ADCPWDNZ = 1;
    AdcaRegs.ADCCTL1.bit.INTPULSEPOS = 1;
    EDIS;

    for(delay_i = 0; delay_i < 200000; delay_i++){}

    EALLOW;
    AdcaRegs.ADCSOC0CTL.bit.CHSEL = 0;      // ADCINA0 (pin 30) -> theta
    AdcaRegs.ADCSOC0CTL.bit.ACQPS = 14;
    AdcaRegs.ADCSOC0CTL.bit.TRIGSEL = 0;
    AdcaRegs.ADCINTSEL1N2.bit.INT1SEL = 0;
    AdcaRegs.ADCINTSEL1N2.bit.INT1E = 1;
    EDIS;

    // CONFIGURACIÓN DAC
    EALLOW;
    CpuSysRegs.PCLKCR16.bit.DAC_B = 1;
    DacbRegs.DACCTL.bit.DACREFSEL = 1;
    DacbRegs.DACCTL.bit.LOADMODE = 0;
    DacbRegs.DACOUTEN.bit.DACOUTEN = 1;
    DacbRegs.DACVALS.all = 2048;
    EDIS;

    for(delay_i = 0; delay_i < 10000; delay_i++){}

    // CONFIGURACIÓN TIMER 0 (10 us)
    InitCpuTimers();
    ConfigCpuTimer(&CpuTimer0, 200, 10); // 10 us

    EALLOW;
    PieVectTable.TIMER0_INT = &cpu_timer0_isr;
    EDIS;

    PieCtrlRegs.PIEIER1.bit.INTx7 = 1;
    IER |= M_INT1;
    EINT;
    ERTM;
    CpuTimer0Regs.TCR.bit.TSS = 0; // Arranca el Timer

    // ==========================================================
    // BUCLE PRINCIPAL (Envía trama con 3 variables)
    // ==========================================================
    for(;;) {
        if(flag_enviar == 1) {
            int pos_scaled = (int)(theta * 1000.0f);
            int vel_scaled = (int)(theta_dot_filt * 1000.0f);
            int u_scaled = (int)(u * 1000.0f);

            char trama[50];
            int len = sprintf(trama, "%d,%d,%d\r\n", pos_scaled, vel_scaled, u_scaled);

            int i;
            for(i = 0; i < len; i++) {
                while (SciaRegs.SCICTL2.bit.TXRDY == 0) {}
                SciaRegs.SCITXBUF.all = trama[i];
            }

            flag_enviar = 0;
        }
    }
}

__interrupt void cpu_timer0_isr(void) {

    // Limpiar bandera de interrupción del ADC y forzar conversión
    AdcaRegs.ADCINTFLGCLR.bit.ADCINT1 = 1;
    AdcaRegs.ADCSOCFRC1.bit.SOC0 = 1;

    // Esperar a que la conversión termine
    volatile int timeout = 1000;
    while((AdcaRegs.ADCINTFLG.bit.ADCINT1 == 0) && (timeout > 0)) {
        timeout--;
    }
    AdcaRegs.ADCINTFLGCLR.bit.ADCINT1 = 1;

    // 1. Leer y escalar la posición medida desde el ADC
    uint16_t adc_raw = AdcaResultRegs.ADCRESULT0;
    theta_measured = THETA_MIN + ((float)adc_raw / 4095.0f) * THETA_RANGE;

    // --- 2. CÁLCULO DE CONTROL (PID) ---
    float error = 0.0f - theta_measured;

    // Integrador con anti-windup
    error_integral += error * Ts;
    if (error_integral > 2.0f)  error_integral = 2.0f;
    if (error_integral < -2.0f) error_integral = -2.0f;

    // --- CÁLCULO DE DERIVADA LENTA (Ts=10 ms) ---
    contador_derivada++;
    if (contador_derivada >= 1000) {
        theta_dot_meas = (theta_measured - theta_prev_slow) / 0.01f;
        theta_prev_slow = theta_measured;
        contador_derivada = 0;
        // Filtro IIR de primer orden, fc ≈ 10 Hz, Ts = 10 ms → alpha = 0.386
        theta_dot_filt = (1.0f - 0.386f) * theta_dot_filt + 0.386f * theta_dot_meas;
    }

    // Calcular acción de control PID
    u = (KP * error) - (KD * theta_dot_filt) + (KI * error_integral);

    // Saturación de la acción de control
    if (u > U_MAX) u = U_MAX;
    if (u < U_MIN) u = U_MIN;

    // --- 3. SIMULACIÓN DE LA PLANTA (Euler-Cromer) ---
    float theta_dot_new = theta_dot + Ts * (gL * theta - (b_fric * invJ * theta_dot) + invJ * u);
    float theta_new = theta + Ts * theta_dot_new; // Euler-Cromer usa la velocidad nueva

    // Saturación del estado de la planta
    if (theta_new > THETA_MAX) theta_new = THETA_MAX;
    if (theta_new < THETA_MIN) theta_new = THETA_MIN;

    theta = theta_new;
    theta_dot = theta_dot_new;

    // --- 4. SALIDA AL DAC (Loopback) ---
    float theta_frac = (theta - THETA_MIN) / THETA_RANGE;
    if(theta_frac < 0.0f) theta_frac = 0.0f;
    if(theta_frac > 1.0f) theta_frac = 1.0f;
    DacbRegs.DACVALS.all = (uint16_t)(theta_frac * 4095.0f);

    // --- 5. BANDERA DE ENVÍO SERIAL CADA 10 ms ---
    contador_serial++;
    if(contador_serial >= 1000) {
        flag_enviar = 1;
        contador_serial = 0;
    }

    PieCtrlRegs.PIEACK.all = PIEACK_GROUP1;
}
